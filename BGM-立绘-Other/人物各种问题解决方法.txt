;人物特效丢失解决方法
;━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
人物特效使用较多，请调整Mugen的各个参数设置以解决特效丢失

还有一种丢失特效的原因是跳过了开场，导致特效丢失，攻击丢失。所以禁止跳过人物开场。

请在Date文件夹里的mugen.cfg中搜索ExlodMax、Helper、MaxWavChannels等参数并调整设置

参考设置:

[Config]

LayeredSpriteMax = 5000

ExplodMax = 2000		主要是调节这个特效最大数

SysExplodMax = 256

HelperMax = 56

GameSpeed = 60

AfterImageMax = 64

LayeredSpriteMax = 20000

PlayerProjectileMax = 10000

WavChannels = 32
